    2. {{- range $i, $n := $path.nodes }}{{ with partial "home/page.html" $n.href }}
  2. {{ partial "home/path-box.html" (dict "icon" $n.icon "label" (index $n $lang | default .Title) "note" (index $n.note $lang) "href" .RelPermalink "group" (i18n "home.path.group")) }}
    3. {{- end }}{{ end }}
  3. {{ partial "home/path-box.html" (dict "icon" $path.client.icon "label" $cl "note" (index $path.client.note $lang) "edge" true) }}
