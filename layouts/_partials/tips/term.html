{{- /* Ссылка на термин глоссария со слоем-определением. Вход: dict "page" "id" "text" (необязательно) "class" (по умолчанию term) "pos" (для сообщений) "icon" (необязательно). Определение накапливается в Scratch страницы и выводится слоем в конце main (дедупликация по id). Неизвестный id ломает сборку. */ -}} {{- $p := .page -}} {{- $id := .id -}} {{- $lang := $p.Language.Lang -}} {{- $e := index site.Data.glossary $id -}} {{- if not $e }}{{ errorf "term: неизвестный id %q (%s); добавьте его в data/glossary.yaml через руководителя документации" $id .pos }}{{ end -}} {{- $t := index $e $lang -}} {{- $text := .text | default $t.term -}} {{- $base := "" -}} {{- with site.GetPage "glossary" }}{{ $base = .RelPermalink }}{{ else }}{{ $base = relLangURL "glossary/" }}{{ end -}} {{- $href := printf "%s#%s" $base $id -}} {{- $tid := printf "tip-t-%s" $id -}} {{- $tips := $p.Scratch.Get "tips" | default dict -}} {{- if not (index $tips $tid) -}} {{- $html := printf `

%s

%s

%s
` $tid ($t.term | htmlEscape) $tid ($t.def | htmlEscape) $href (i18n "term.more") -}} {{- $p.Scratch.SetInMap "tips" $tid ($html | safeHTML) -}} {{- end -}} {{- $p.Scratch.Set "f-tips" true -}} {{ with .icon }}{{ partial "ui/icon.html" . }}{{ end }}{{ $text }} {{- /* Без пробела после тега: стоит в потоке текста. */ -}}
