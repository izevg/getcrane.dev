{{- /* Спрайт Lucide: из дословных файлов assets/icons/svg, один отпечатанный файл на сайт (кэшируется между страницами). Выход: адрес спрайта. Вызывать через partialCached. */ -}} {{- $parts := slice -}} {{- range resources.Match "icons/svg/*.svg" -}} {{- $name := strings.TrimSuffix ".svg" (path.Base .Name) -}} {{- $inner := replaceRE `(?s)^.*?]*>\s*(.*?)\s*\s*$` "$1" .Content -}} {{- $parts = $parts | append (printf `%s` $name $inner) -}} {{- end -}} {{- $sprite := resources.FromString "icons/sprite.svg" (printf `` (delimit $parts "")) | fingerprint -}} {{- return $sprite.RelPermalink -}}
