{{- /* {{< term "id" >}} или {{< term "id" "текст в нужной форме" >}}. Ссылка на //glossary/#id; определение рендерится слоем в конце main (дедупликация по id). Неизвестный id ломает сборку. */ -}} {{- $id := .Get 0 -}} {{- $p := .Page -}} {{- $lang := $p.Language.Lang -}} {{- $e := index site.Data.glossary $id -}} {{- if not $e }}{{ errorf "term: неизвестный id %q (%s); добавьте его в data/glossary.yaml через руководителя документации" $id .Position }}{{ end -}} {{- $t := index $e $lang -}} {{- $text := .Get 1 | default $t.term -}} {{- $base := "" -}} {{- with site.GetPage "glossary" }}{{ $base = .RelPermalink }}{{ else }}{{ $base = relLangURL "glossary/" }}{{ end -}} {{- $href := printf "%s#%s" $base $id -}} {{- $tid := printf "tip-t-%s" $id -}} {{- $tips := $p.Scratch.Get "tips" | default dict -}} {{- if not (index $tips $tid) -}} {{- $html := printf `

%s

%s

%s
` $tid ($t.term | htmlEscape) $tid ($t.def | htmlEscape) $href (i18n "term.more") -}} {{- $p.Scratch.SetInMap "tips" $tid ($html | safeHTML) -}} {{- end -}} {{- $p.Scratch.Set "f-tips" true -}} {{ $text }} {{- /* Без пробела после тега: шорткод стоит в потоке текста. */ -}}
