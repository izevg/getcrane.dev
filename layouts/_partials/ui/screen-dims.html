{{- /* Размеры снимка Storybook из data/screenshot-variants.yaml. Вход: dict "story" "variant". Сначала точный ключ «История/Вариант» (S9Journal/DesktopDark), затем самое длинное совпадение по началу имени варианта (Desktop1440Sheet подходит к ключу Desktop1440). Выход: dict width, height; нет описания — сборка падает. */ -}} {{- $dims := dict -}} {{- $table := index site.Data "screenshot-variants" -}} {{- with index $table (printf "%s/%s" .story .variant) }}{{ $dims = . }}{{ else }} {{- $best := "" -}} {{- range $k, $v := $table }}{{ if and (not (strings.Contains $k "/")) (hasPrefix $.variant $k) (gt (len $k) (len $best)) }}{{ $best = $k }}{{ $dims = $v }}{{ end }}{{ end -}} {{- end -}} {{- if not $dims.width }}{{ errorf "screens: снимок %s/%s не описан в data/screenshot-variants.yaml" .story .variant }}{{ end -}} {{- return $dims -}}
