{{- /* Диаграмма mermaid. SVG (светлый и тёмный) рендерит scripts/render-mermaid.ts при сборке в CI → assets/diagrams/<хэш>.{light,dark}.svg; у читателя JS для диаграмм нет. Хэш — первые 16 знаков SHA-256 от исходника без краевых пробелов (тот же расчёт в скрипте). Локально шаг пропускается переменной CRANE_SKIP_MERMAID=1; тогда вместо картинки выводится исходник (в production без SVG — ошибка сборки). Подпись обязательна: ```mermaid {caption="…"}. */ -}} {{- $src := trim .Inner " \n\r\t" -}} {{- $h := substr (crypto.SHA256 $src) 0 16 -}} {{- $light := resources.Get (printf "diagrams/%s.light.svg" $h) -}} {{- $dark := resources.Get (printf "diagrams/%s.dark.svg" $h) -}} {{- $cap := .Attributes.caption -}} {{- if not $cap }}{{ warnidf "mermaid-caption" "mermaid: у диаграммы нет подписи (caption) в %s" .Page.Path }}{{ end -}} {{- $capid := printf "dg-%s-cap" $h -}}
