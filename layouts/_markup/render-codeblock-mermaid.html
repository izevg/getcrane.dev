{{- /* Диаграмма mermaid. SVG (светлый и тёмный) рендерит scripts/render-mermaid.ts локально (npm run mermaid) → assets/diagrams/<хэш>.{light,dark}.svg; в CI рендера нет, scripts/check-mermaid.ts проверяет наличие и актуальность файлов. У читателя JS для диаграмм нет. Хэш — первые 16 знаков SHA-256 от исходника без краевых пробелов (тот же расчёт в скрипте). Без SVG сборка в production падает. Подпись обязательна: ```mermaid {caption="…"}. Целевая топология: {caption="…" state="target"} — бейдж «Целевое состояние» в начале подписи. Структура: figure → рамка → регион прокрутки (role и tabindex ставит остров при переполнении) → обёртка role=img → inline-SVG. */ -}} {{- $src := trim .Inner " \n\r\t" -}} {{- $h := substr (crypto.SHA256 $src) 0 16 -}} {{- $light := resources.Get (printf "diagrams/%s.light.svg" $h) -}} {{- $dark := resources.Get (printf "diagrams/%s.dark.svg" $h) -}} {{- $cap := .Attributes.caption -}} {{- $state := .Attributes.state -}} {{- if and $state (ne $state "target") (ne $state "current") }}{{ errorf "mermaid: state=%q: допустимы target и current (%s)" $state .Page.Path }}{{ end -}} {{- if not $cap }}{{ warnidf "mermaid-caption" "mermaid: у диаграммы нет подписи (caption) в %s" .Page.Path }}{{ end -}} {{- $capid := printf "dg-%s-cap" $h -}} {{- $width := 0 -}} {{- with $light }}{{ with findRESubmatch `viewBox="[-\d.]+ [-\d.]+ ([\d.]+) [\d.]+"` .Content 1 }}{{ $width = int (float (index (index . 0) 1)) }}{{ end }}{{ end -}}
